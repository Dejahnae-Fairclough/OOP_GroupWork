#ifndef Product_h
#define Product_h
#include <iostream>
#include <iostream>
#include <string>
#include <cstring>

class Product
{
protected:
	int id;
	char* name;
	double price;

public:
	Product() {
		this->id = 0;
		this->name = (char*)"";
		this->price = 0;
	}

	Product(int id, char* name, double price){

		this->id = id;
		this->name = name;
		this->price = price;
	
	}

	Product(Product* xProduct){
		this->id = xProduct->id;
		this->name = xProduct->name;
		this->price = xProduct->price;
	
	}

	int getID() {
		return id;
	}
	char* getName() {
		return name;
	}
	double getPrice() {
		return price;
	}

	void setID(int id) {
		this->id = id;
	}
	void setName(char* name) {
		this->name = name;
	}
	void setPrice(double price) {
		this->price = price;
	}
	virtual ~Product() {};

	virtual void taxGCT() = 0;

	virtual void Display() = 0;

	 
	


};

#endif
